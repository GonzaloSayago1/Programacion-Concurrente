Monitor administradorFotocopias
{
    int esperando = 0;
    int libre = 10;
    cola maquinasLibres; // Asumo que se inicializa con IDs{0..9}

    cond hayTrabajo;
    cond esperaTurno;
    cond esperaDato;

    int maquinaAsignada; //Variable temporal para dato

    procedure solicitarTurno(mi_maq: out int)
    {
        esperando++;
        signal(hayTrabajo);
        wait(esperaTurno);

        //Cuando la persona despierta, lee el dato y le avisa al empleado
        mi_maq = maquinaAsignada;
        signal(esperaDato); //Avisa que leyo dato para que el empleado pueda seguir asignando turnos
    }

    procedure llamarSiguiente()
    {
        if(esperando == 0 || libre == 0)
            wait(hayTrabajo);
        esperando--;
        libre--;
        int sig;
        pop(maquinasLibres, sig); //Saca fotocopiadora libre

        //Deja el dato y espera que la persona lo lea
        maquinaAsignada = sig;
        signal(esperaTurno); //Despierta persona
        wait(esperaDato) //Espera que la persona lea el dato
    }

    procedure notificarFin(mi_maq: in int)
    {
        push(maquinasLibres, mi_maq);
        libres++;
        signal(hayTrabajo); // Avisa al empleado que hay una maquina libre
    }
}

process persona [id: 0..N-1]
{
    int fotocopiadora;
    administradorFotocopias.solicitarTurno(fotocopiadora);
    Fotocopiar(fotocopiadora);
    administradorFotocopias.notificarFin(fotocopiadora);
}

process Empleado
{
    for i: 0 to N-1
    {
        administradorFotocopias.llamarSiguiente();
    }
}
