Nota: El JTP al no tener comportamiento por fuera del monitor, deberia ser un monitor.

monitor aula
{
    int cant = 0;
    cond barrera;

    procedure llegada(grupo: out int)
    {
        cant++;
        if (cant < 50)
        {
            wait(barrera);
        }
        else
        {
            signal_all(barrera); // El último en llegar libera a los otros 49
        }

        grupo = AsignarNroGrupo();
    }
}

Monitor jtp
{
    int puntaje = 25;
    entregaGrupo[1..25] = ([25] 0);
    cond esperaNota[grupo];
    notaDelGrupo[1..25];


    procedure asignarNota(grupo: in int; nota: out int)
    {
        entregaGrupo[grupo]++;
        if(entregaGrupo[grupo] == 2)
        {
            notaDelGrupo[grupo] = puntaje;
            puntaje--;
            signal(esperaNota[grupo]); //Despierto a mi compañero que estaba esperando
        }
        else
        {
            wait(esperaNota[grupo]);
        }
        //Recibe su nota y se va
        nota = notaDelGrupo[grupo];
    }
}

process alumno[id: 0..49]
{
    int numGrupo, notaGrupo;

    aula.llegada(id, numGrupo);
    //Realiza tarea
    jtp.asignarNota(numGrupo, notaGrupo);
}
