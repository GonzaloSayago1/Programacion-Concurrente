monitor Maraton
{
    int cantidad = 0;
    cond esperaLlegada;

    procedure llegada()
    {
        cantidad++;
        if (cantidad < C)
        {
            wait(esperaLlegada);
        }
        else
        {
            signal_all(esperaLlegada);
        }
    }
}

monitor FilaMaquina
{
    int botellas = 20;
    bool ocupada = false;
    int esperando = 0;
    cond fila;

    procedure tomarTurno(hayBotellas: out bool)
    {
        if (ocupada || esperando > 0)
        {
            esperando++;
            wait(fila);
        }
        else
        {
            ocupada = true;
        }

        hayBotellas = (botellas > 0);
        if (hayBotellas)
        {
            botellas--; // Si hay stock, toma su botella directamente
        }
    }

    procedure reponerYTomar()
    {
        botellas = 20;
        botellas--; // El corredor actual saca su botella
    }

    procedure liberarTurno()
    {
        if (esperando > 0)
        {
            esperando--;
            signal(fila);     // Pasa la máquina al siguiente en la fila FIFO
        }
        else
        {
            ocupada = false;  // No hay nadie esperando, queda libre
        }
    }
}

monitor CoordinacionReposicion
{
    bool hayAviso = false;
    cond esperaRepositor;
    cond reposicionLista;

    procedure avisarYEsperar()
    {
        hayAviso = true;
        signal(esperaRepositor);
        wait(reposicionLista);    // Espera a que termine la carga física
    }

    procedure esperarAviso()
    {
        if(not hayAviso){wait(esperaRepositor)}
        hayAviso = false; //Resetea variable auxiliar
    }

    procedure notificarFin()
    {
        signal(reposicionLista);  // Le avisa al corredor que ya terminó
    }
}

process Corredor [id: 0..C-1]
{
    bool hayBotellas;

    Maraton.llegada();
    // Corre maraton
    FilaMaquina.tomarTurno(hayBotellas);

    if (!hayBotellas)
    {
        CoordinacionReposicion.avisarYEsperar(); //Llama al repositor
        FilaMaquina.reponerYTomar(); // Avisa que la maquina se lleno nuevamente
    }

    FilaMaquina.liberarTurno();
}

process Repositor
{
    while (true)
    {
        CoordinacionReposicion.esperarAviso();
        // Carga físicamente las 20 botellas (tarea fuera de monitores)
        CoordinacionReposicion.notificarFin();
    }
}
