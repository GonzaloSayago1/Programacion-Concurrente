sem disponible = 1;

process Persona [id: 0..N-1]
{
    P(disponible)
    Imprimir(documento)
    V(disponible)
}



B)
sem colaDisponible = 1;
bool impresoraDisponible = true;
sem espera[N] = ([N] 0)
queue cola;

process Persona [id: 0..N-1]
int sig;
{
    P(colaDisponible)

    if(impresoraDisponible)//Si la impresora esta disponible, libero la cola
    {
        impresoraDisponible = false;
        V(colaDisponible)
    }
    else
    {
        push(cola, id)
        V(colaDisponible)
        P(espera[id])//Espera hasta que lo despierten
    }

    Imprimir(documento)

    P(colaDisponible)
    if(empty(cola))
    {
        impresoraDisponible = true; //Si no hay nadie esperando por usar la impresora, la libero
    }
    else
    {
        sig = pop(cola)
        V(espera[sig])//Despierto al siguiente
    }
    V(colaDisponible)
}




C)
sem turno[N] = ([N] 0);

process Persona[id: 0..N-1]
{
    //El primer proceso con id 0 se ejecuta primero
    if(id > 0)
    {
        P(turno[id])//Espera su turno
    }

    Imprimir(documento)

    if(id < N-1)
    {
        V(turno[id + 1]);
    }
}




D)
sem colaDisponible = 1;
sem hayAlgo = 0;
sem impresoraDisponible = 1;
sem espera[N] = ([N] 0);
cola C;

process Persona [id: 0..N-1]
{
    // Llega y se encola de forma protegida
    P(colaDisponible);
    push(C, id);
    V(colaDisponible);
    V(hayAlgo); // Avisa al coordinador que llegó alguien

    // Espera a que el coordinador le ceda el turno
    P(espera[id]);

    // Usa la impresora
    Imprimir(documento);

    // Libera el recurso de impresión
    V(impresoraDisponible);
}

process Coordinador
{
    int sig;

    for [i = 0 to N-1]
    {
        P(hayAlgo); // Espera a que haya alguien en la cola

        P(colaDisponible);
        pop(C, sig);
        V(colaDisponible);

        P(impresoraDisponible); // Espera a que la impresora esté desocupada
        V(espera[sig]); // Despierta a la persona correspondiente
    }
}



E)
sem colaDisponible = 1, hayAlgo = 0, espera[N] = ([N] = 0);
//Control de impresoras
sem mutexImpresoras = 1, cantImpresoras = 5;
cola impresorasLibres, colaLLegada;
int impresoraAsignada[N];

process Persona [id: 0..N-1]
{
    //Llega y se encola
    P(colaDisponible)
    Push(colaLlegada, id)
    V(colaDisponible)
    V(hayAlgo) //Avisa que llego

    //Espera que lo despierten
    P(espera[id])

    //Ve que impresora le fue asignada y la usa
    int miImpresora = impresoraAsignada[id]
    Imprimir(documento, miImpresora)

    //Libera la impresora que usó
    P(mutexImpresoras)
    Push(impresorasLibres, miImpresora)
    V(mutexImpresoras)

    V(cantImpresoras) //Avisa que hay una impresora libre más
}

process Coordinador
{
    int sig, idImpresora;

    for [i = 0 to N-1]
    {
        P(hayAlgo)
        P(cantImpresoras)

        //Saca persona de la cola
        P(colaDisponible)
        sig = Pop(colaLlegada)
        V(colaDisponible)

        //Saca impresora libre
        P(mutexImpresoras)
        idImpresora = Pop(impresorasLibres)
        V(mutexImpresoras)

        //Asigna a la siguiente persona la impresora que debe usar
        impresoraAsignada[sig] = idImpresora

        //Despierta a la siguiente persona
        V(espera[sig])
    }
}
