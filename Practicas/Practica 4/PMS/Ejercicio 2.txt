process Preparador
{
    text muestra;

    while (true)
    {
        muestra = prepararMuestra();
        Admin!enviarMuestra(muestra);
    }
}

process Armador
{
    text muestra, res;

    while (true)
    {
        //Solicita la siguiente muestra al Admin
        Admin!pedirMuestra();
        Admin?recibirMuestra(muestra);

        //Arma el set y se lo envía al analista
        muestra = armarSet();
        Analista!analisis(muestra);

        //Espera el resultado para archivarlo
        Analista?resultado(res);
        archivar(res);
    }
}

process Analista
{
    text muestra, res;

    while (true)
    {
        //Espera una muestra para analizar
        Armador?analisis(muestra);
        res = analizar(muestra);

        //devuelve el resultado al armador
        Armador!resultado(res);
    }
}

process Admin
{
    cola buffer;
    text muestra;

    do Preparador?enviarMuestra(muestra) ->
        push(buffer, muestra);

    [] not empty(buffer); Armador?pedirMuestra() ->
        pop(buffer, muestra);
        Armador!recibirMuestra(muestra);
    od
}
