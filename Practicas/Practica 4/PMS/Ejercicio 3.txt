{
    text exam, res;

        Admin!pedidos();

        res = corregir(exam);
        Alumno[idAlumno]!resultados(res);
    }
}

process Admin
{
    text exam
    int idAlumno;
    cola colaExamenes;
    int cantEntregas = 0;

    do cantEntregas < N; Alumno[*]?entregas(exam, idAlumno) ->
        push(colaExamenes, (exam, idAlumno));
        cantEntregas++;
    [] not empty(cola); profesor?pedidos() ->
        pop(colaExamenes, (exam, idAlumno));
        Profesor!examenes(exam, idAlumno);
    od
}

process Profesor [id: 0..P-1]
{
    text exam, res;
    int idAlumno;

    Admin!pedidos(id);
    Admin?examenes(exam, idAlumno);

    while (idAlumno != -1)
    {
        res = corregir(exam);
        Alumno[idAlumno]!resultados(res);

        // Pide el siguiente examen
        Admin!pedidos(id);
        Admin?examenes(exam, idAlumno);
    }
}

process Admin
{
    text exam;
    int idAlumno, idProfesor;
    cola colaExamenes;
    int cantEntregas = 0;

    do cantEntregas < N; Alumno[*]?entregas(exam, idAlumno) ->
        push(colaExamenes, (exam, idAlumno));
        cantEntregas++;

    [] not empty(colaExamenes); Profesor[*]?pedidos(idProfesor) ->
        pop(colaExamenes, (exam, idAlumno));
        Profesor[idProfesor]!examenes(exam, idAlumno);
    od

    // Bloque de finalización
    for i: 1 to P
    {
        Profesor[*]?pedidos(idProfesor);

    }
}

process Profesor [id: 0..P-1]
{
    text exam, res;
    int idAlumno;

    Admin!pedidos(id);
    Admin?examenes(exam, idAlumno);

    while (idAlumno != -1)
    {
        res = corregir(exam);

        // Pide el siguiente examen
        Admin!pedidos(id);
        Admin?examenes(exam, idAlumno);

process Admin
{
    text exam;
    int idAlumno, idProfesor;
    cola colaExamenes;
    int cantEntregas = 0;

    for i: 1 to N
    {
        Alumno[*]?llegada();
    }

    for i: 1 to N
    {
        Alumno[i-1]!espera();
    }

    do cantEntregas < N; Alumno[*]?entregas(exam, idAlumno) ->
        push(colaExamenes, (exam, idAlumno));
        cantEntregas++;

    [] not empty(colaExamenes); Profesor[*]?pedidos(idProfesor) ->
        pop(colaExamenes, (exam, idAlumno));
        Profesor[idProfesor]!examenes(exam, idAlumno);
    od

    // Bloque de finalización
    for i: 1 to P
    {
        Profesor[*]?pedidos(idProfesor);
        Profesor[idProfesor]!examenes("", -1);
    }
}
