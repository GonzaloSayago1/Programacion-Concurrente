process Coordinador
{
    int idImp;
    text doc;

    for i: 1 to (N * 10)
    {
        receive solicitudes(doc);
        receive disponibles(idImp);
        send siguiente[idImp](doc);
    }

    // Espera a que cada una de las 3 impresoras pida trabajo y le manda terminar
    for i: 1 to 3
    {
        receive disponibles(idImp);
        send siguiente[idImp]("TERMINAR");
    }
}

process impresora[id: 0..2]
{
    text doc = "";

    while(doc != "TERMINAR")
    {
        send disponibles(id);
        receive siguiente[id](doc);

        if(doc != "TERMINAR") {imprimir(doc)}
    }
}
