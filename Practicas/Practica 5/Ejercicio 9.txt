procedure Recoleccion is
    Task type Camion is
        identificador(id: in integer);
    end Camion;

    Task type Persona is
        Entry reclamo;
        Entry identificador(id: in integer);
    end Persona;

    Task Administrador is
        Entry camionLibre(id: in integer);
        Entry sumarReclamo(id: in integer);
    end Administrador;

    arrPersona: array (1..P) of Persona;
    arrCamion: array (1..3) of Camion;

    Task body Camion is
        id:integer;
    begin
        accept identificador (pos:in integer) do
            id:= pos;
        end identificador;
        LOOP
            //Solicita trabajo
            Aministrador.camionLibre(id, idReclamo);
            arrPersona(idReclamo).reclamo;
        end LOOP;
    end Camion;

    Task body Persona is
        aceptado: boolean := false;
        id: integer;
    begin
        accept identificador (pos: in integer) do
            id:= pos;
        end identificador;
        while (not aceptado) loop
            Administrador.sumarReclamo(id);
            SELECT
                accept reclamo;
                aceptado:= true;
            OR DELAY 900.0
                null;
            end SELECT;
        end loop;
    end Persona;

    Task body Administrador is
        posMax: integer
        reclamos: array(1..P) of integer := 0; //Contador de reclamos
        esperando: integer := 0;
    begin
        loop
            SELECT
                WHEN (esperando > 0) => accept camionLibre(id: in integer; idReclamo: out integer) do
                    posMax:= obtenerPosicionMaxima(reclamos);
                    idReclamo:= posMax; //Rendezvous
                    reclamos(posMax):= -1; //De esta forma no lo voy a tener más en cuenta
                    esperando:= esperando - 1; //Decrementa una persona esperando

                end camionLibre;
            OR
                accept sumarReclamo(id: in integer) do
                    if (reclamos(i) = 0)
                        esperando:= esperando + 1;
                    end if;
                    reclamos(i):= reclamos(i) + 1;
                end sumarReclamo;
            end SELECT;
        end loop;

    end Administrador;

begin
    for i in 1..3 loop
        arrCamion(i).identificador(i);
    end loop;

    for i in 1..P loop
        arrPersona(i).identificador(i);
    end loop;
end Recoleccion;
