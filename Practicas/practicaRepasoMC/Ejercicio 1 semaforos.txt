        V(mutexTerminal);

        push(cola, id);
        V(mutexCola);
        P(espera[id]);
    }
    //Usa terminal
    usarTerminal();
    //Llama al siguiente o libera la maquina
    P(mutexCola);

    {
        P(mutexTerminal)
        terminalLibre = true;
        V(mutexTerminal);
    }
    else {
        //Despierta al siguiente
        pop(cola, siguiente);
        V(espera[siguiente]);
    }
    V(mutexCola);
}
