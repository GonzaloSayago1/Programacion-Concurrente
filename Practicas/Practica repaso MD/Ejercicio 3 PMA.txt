chan avisos(boolean);
chan atencionEmbarazada(text, real, int);
chan atencionMenosBoletas(text, real, int);
chan atencionGeneral(text, real, int);
chan atencion[P](real, text); //Asumo que va de 1 a P

process Persona[id: 1..P]
{
    int cantidadBoletas = determinarCantidad();
    boolean esEmbaradaza;
    text boleta, recibo;
    real pago, vuelto;

    pago = determinarPago(boleta);

    if(esEmbaradaza){
        send atencionEmbarazada(boleta, pago, id);
    }
    else if(cantidadBoletas < 5) {
        send atencionMenosBoletas(boleta, pago, id);
    }
    else {
        send atencionGeneral(boleta, pago, id);
    }

    //Avisa que hay algo esperando
    send avisos(true);

    //Espera atencion
    receive atencion[id](vuelto, recibo);
}

process Administrador
{
    text boleta, recibo;
    real pagoRecibido, vuelto;
    int idPersona;

    while (true)
    {
        receive avisos(ok);

        if(not empty(atencionEmbarazada)){
            receive atencionEmbarazada(boleta, pagoRecibido, idPersona);
        }
        else if(not empty(atencionMenosBoletas)){
            receive atencionMenosBoletas(boleta, pagoRecibido, idPersona);
        }
        else{
            receive atencionGeneral(boleta, pagoRecibido, idPersona);
        }

        vuelto = determinarVuelto(boleta, pagoRecibido);
        recibo = generarRecibo(boleta);

        send atencion[idPersona](vuelto, recibo);
    }
}
