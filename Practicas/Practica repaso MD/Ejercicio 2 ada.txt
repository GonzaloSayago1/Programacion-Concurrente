procedure Negocio is
    Task type Persona;

    Task Administrador is
        Entry atencionAncianos(boleta: in text; pago: in float; vuelto: out float; rec: our text);
        Entry atencionMenosBoletas(boleta: in text; pago: in float; vuelto: out float; rec: our text);
        Entry atencionGeneral(boleta: in text; pago: in float; vuelto: out float; rec: our text);
    end Administrador;

    arrPersonas: array (1..P) of Persona;

    Task body Persona is
        cantBoletas: integer;
        esJubilado: boolean;
        boleta: text; //Asumo que esta inicializado
        pago, vuelto: float;
        recibo: text;
    begin
        cantBoletas:= determinarCantidad();
        esJubilado:= determinarSiEsJubilado();

        if(esJubilado)
            Administrador.atencionAncianos(boleta, pago, vuelto, recibo);
        elseif(catBoletas < 5)
            Administrador.atencionGeneral(boleta, pago, vuelto, recibo);
        else
            Administrador.atencionGeneral(boleta, pago, vuelto, recibo);
        end if;
    end Persona;

    Task body Administrador is
        montoBoleta: float;
    begin
        for i in 1..P loop
            SELECT
                accept atencionAncianos(boleta: in text; pago: in float; vuelto: out float; rec: out text) do
                    rec := generarRecibo(boleta);
                    montoBoleta := obtenerMonto(boleta);
                    vuelto := pago - montoBoleta; //Asumo que no van a dar menos plata que el monto de la boleta
                end atencionAncianos;
            OR when (atencionAncianos'count = 0) =>
                acccept atencionMenosBoletas(boleta: in text; pago: in float; vuelto: out float; rec: out text) do
                    rec := generarRecibo(boleta);
                    montoBoleta := obtenerMonto(boleta);
                    vuelto := pago - montoBoleta; //Asumo que no van a dar menos plata que el monto de la boleta
                end atencionMenosBoletas;
            OR when (atencionAncianos'count = 0) AND (atencionMenosBoletas'count = 0)
                accept atencionGeneral(boleta: in text; pago: in float; vuelto: out float; rec: out text) do
                    rec := generarRecibo(boleta);
                    montoBoleta := obtenerMonto(boleta);
                    vuelto := pago - montoBoleta; //Asumo que no van a dar menos plata que el monto de la boleta
                end atencionGeneral;
            end SELECT;
        end loop;
    end Administrador;

begin
    null;
end Negocio;
